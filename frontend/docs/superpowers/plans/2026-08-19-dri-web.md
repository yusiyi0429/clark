# DRI 侧网页 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在同一个应用里开出第二道门，让 Domain DRI 能接收并逐条调整 OM、接收调研议题任务、上报信号，且服务端真的分得清是谁在调用。

**Tech Stack:** Next.js 16.3 App Router（`middleware` 已改名 `proxy`）· React 19.2 · TypeScript 5.9 strict · Tailwind 4 · SWR · vitest · pnpm。落盘是 `.data/store.json` 整库 JSON 快照（`src/lib/data/persist.ts`），单进程单写者。

设计原件：[docs/superpowers/specs/2026-08-19-dri-web-design.md](../specs/2026-08-19-dri-web-design.md)。本计划的每一节都可以回溯到 spec 的某一节，节号写在任务标题后面。

`src/lib/identity.ts`：

```ts
import { createHmac, timingSafeEqual } from "node:crypto";

/**
 * DRI 的身份 cookie。和 `src/lib/access.ts` 那把共享口令是两件事：
 * 口令回答「你能不能进这个演示环境」，这个回答「你是谁」。
 *
 * 共用 `TKOS_ACCESS_SECRET` 是有意的 —— 不为六个内部人多加一个环境变量。
 * 代价写在 spec §4.1：撤销一条链接只能轮换这个密钥，而那会同时把一号位的
 * 访问 cookie 一起踢掉。等外部人（陪跑、灯塔）要进来必须换成带 kid 的签名
 * 加一份吊销名单，那时候要动的只有这个文件。
 */
export const WHO_COOKIE_NAME = "tkos_who";

/** 30 天。一个月一次的 OM 周期，比这短就等于每期都要重发链接。 */
export const WHO_COOKIE_MAX_AGE = 60 * 60 * 24 * 30;

/** personId 会被拼进 URL 和 cookie，也会被拿去查对象库 —— 只放 id 里真实存在的字符。 */
const ID_SHAPE = /^[A-Za-z0-9_-]+$/;

function configuredSecret(): string | null {
  return process.env.TKOS_ACCESS_SECRET?.trim() || null;
}

function sameText(left: string, right: string): boolean {
  const leftBytes = Buffer.from(left);
  const rightBytes = Buffer.from(right);
  return leftBytes.length === rightBytes.length && timingSafeEqual(leftBytes, rightBytes);
}

/**
 * 签一条。**token 自己带着 personId**（`PSN-06.<mac>`）—— 因为收件那一侧要从
 * cookie 里知道是谁，而不是拿一份名册去逐个试签名。
 */
export function signPersonId(personId: string): string | null {
  const secret = configuredSecret();
  if (!secret) return null;

  const id = personId.trim();
  if (!ID_SHAPE.test(id)) return null;

  const mac = createHmac("sha256", secret).update(`tkos-who-v1:${id}`).digest("base64url");
  return `${id}.${mac}`;
}

/** 验一条。解出 personId 或 null。**任何失败都是 null，不抛** —— 它跑在 proxy 里。 */
export function personIdFromToken(token: string | undefined): string | null {
  if (!token) return null;

  const dot = token.lastIndexOf(".");
  if (dot <= 0) return null;

  const id = token.slice(0, dot);
  const expected = signPersonId(id);
  return expected && sameText(token, expected) ? id : null;
}
```

```ts
import { AsyncLocalStorage } from "node:async_hooks";

/**
 * 这一次请求是谁在调。
 *
 * `role` 只有两档，因为授权只有两档（spec §4.4 的可见性表）。不做角色继承、
 * 不做权限位 —— C-1 已定：只做够用的那一层。
 */
export type Actor = { personId: string; role: "ceo" | "dri" };

/**
 * 为什么是 AsyncLocalStorage 而不是把 actor 当参数传下去：`store.ts` 里 70 处
 * 调用点、40 处 useResource 的签名全靠「一个字不改」才成立（见 client.ts 文件头）。
 * 加一个首参会把这条链路上的每个方法、每个视图一起改掉。
 */
const storage = new AsyncLocalStorage<Actor>();

export function runAsActor<T>(actor: Actor, fn: () => T): T {
  return storage.run(actor, fn);
}

/**
 * 没有作用域时返回 null，**不在这里回落成一号位**。回落是产品判断（vitest、
 * 脚本、后台结算都没有请求身份），归 `store.ts` 的 `viewer()` 决定；这一层
 * 只如实说「这一次没有身份」。
 */
export function currentActor(): Actor | null {
  return storage.getStore() ?? null;
}
```

```ts
/**

const viewer = (): Person => {
  const id = currentActor()?.personId;
  const found = id ? tk.people.find((p) => p.id === id) : undefined;
  return found ? found : findPerson(tk.VIEWER_ID);
};
```

```ts
export const CEO_ACTOR: Actor = { personId: tk.VIEWER_ID, role: "ceo" };
```

```ts
 * 数据层的唯一入口。前端 `lib/data/client.ts` 的方法全部打到这里。
 *
 * 为什么是一个 RPC 端点而不是几十条 REST 路由：这不是公开 API，只被我们自己的
 * 前端调用。1:1 映射能让 client.ts 的调用点真正做到一个字不改 —— 那正是
 * client.ts 文件头承诺的「只改方法体」。铺几十条路由只是把同一件事写几十遍。
 *
 * 安全边界三层，缺一不可：
 *
 * 1. 方法名必须命中白名单（白名单就是 `api` 自己的键）。不接受任意属性路径，不 eval。
 * 2. **身份**：一号位那把共享口令 → 一号位；`tkos_who` 签名 cookie → 解出来的那个
 *    DRI。两个都没有 → 401。（`src/proxy.ts` 已经挡过一次，这里不靠它 ——
 *    proxy 只验签，认不认人是数据层的事。）
 * 3. **可见性**：`METHOD_SCOPE` 查一下这个身份能不能调这个方法。少了这一层，
 *    一条 DRI cookie 就能调 `signClosure`。
 */
import { cookies } from "next/headers";
import { NextResponse } from "next/server";

import { ACCESS_COOKIE_NAME, isAccessCookieValid } from "@/lib/access";
import { WHO_COOKIE_NAME, personIdFromToken } from "@/lib/identity";
import type { Actor } from "@/lib/data/actor";
import { runAsActor } from "@/lib/data/actor";
import { scopeOf } from "@/lib/data/scope";
import { CEO_ACTOR, api, persistSoon, resolveDriActor } from "@/lib/data/store";
import { ApiError } from "@/lib/data/shared";

export const dynamic = "force-dynamic";
export const runtime = "nodejs";

type Body = { method?: unknown; args?: unknown };

const METHODS = new Set(
  Object.entries(api)
    .filter(([, v]) => typeof v === "function")
    .map(([k]) => k)
);

/** 一号位优先：两个 cookie 都带着的时候，他那把口令说了算（spec §4.2）。 */
async function actorOf(): Promise<Actor | null> {
  const jar = await cookies();
  if (isAccessCookieValid(jar.get(ACCESS_COOKIE_NAME)?.value)) return CEO_ACTOR;

  const personId = personIdFromToken(jar.get(WHO_COOKIE_NAME)?.value);
  return personId ? resolveDriActor(personId) : null;
}

export async function POST(req: Request) {
  let body: Body;
  try {
    body = (await req.json()) as Body;
  } catch {
    return NextResponse.json({ ok: false, error: "请求体不是合法 JSON。" }, { status: 400 });
  }

  const { method, args } = body;
  if (typeof method !== "string" || !METHODS.has(method)) {
    return NextResponse.json({ ok: false, error: `未知方法：${String(method)}` }, { status: 400 });
  }
  if (args !== undefined && !Array.isArray(args)) {
    return NextResponse.json({ ok: false, error: "args 必须是数组。" }, { status: 400 });
  }

  const actor = await actorOf();
  if (!actor) {
    return NextResponse.json({ ok: false, error: "没有身份。" }, { status: 401 });
  }

  const scope = scopeOf(method);
  if (!scope) {
    // 白名单过了但表里没有 —— 只可能是有人加了方法没进表（scope.test.ts 会红）。
    console.error(`[api/data] ${method} 不在 METHOD_SCOPE 里，按拒绝处理。`);
    return NextResponse.json({ ok: false, error: "这个方法还没有分类。" }, { status: 500 });
  }
  if (scope !== "both" && scope !== actor.role) {
    console.warn(`[api/data] ${actor.personId}（${actor.role}）想调 ${method}（${scope}）—— 拒了。`);
    return NextResponse.json(
      { ok: false, error: "这个动作不在你这一侧。", kind: "forbidden" },
      { status: 403 }
    );
  }

  const fn = (api as unknown as Record<string, (...a: unknown[]) => Promise<unknown>>)[method];

  try {
    // 身份塞进请求作用域：`viewer()` 从这里读，于是每一处署名都是真的。
    const data = await runAsActor(actor, () => fn(...((args ?? []) as unknown[])));
    // 读操作也可能改 store（settleResearch 会把到点的研究装上），所以每次都排队落盘。
    // persist.ts 里做了防抖，连续操作合并成一次写。
    persistSoon();
    return NextResponse.json({ ok: true, data: data ?? null });
  } catch (err) {
    if (err instanceof ApiError) {
      // 业务性失败：把原文带回给前端，它会原样显示。
      return NextResponse.json({ ok: false, error: err.message, kind: "api" }, { status: 409 });
    }
    console.error(`[api/data] ${method} 抛出：`, err);
    return NextResponse.json(
      { ok: false, error: err instanceof Error ? err.message : "服务端错误" },
      { status: 500 }
    );
  }
}
```

```ts
/*
 * `/api/signals/inbox` 与 `/api/om/intake` 在这里，不是因为它们公开，而是因为
 * **推它们的是机器**：抓取作业和灌纲领的那条命令都没有访问 cookie，不加这两条
 * 它们连 route 都进不到，只会拿到 `{"error":"unauthorized"}` —— 而那句话读起来
 * 像 `OPERATOR_TOKEN` 配错了，能查很久。它们自己那道门是请求头里的

 *
 * `/dri/enter` 也在这里：DRI 没有那把共享口令，这是他进来的唯一入口，它自己的门
 * 是 URL 里那道签名（`src/lib/identity.ts`）。
 */
const PUBLIC_PATHS = new Set([
  "/access",
  "/api/access",
  "/api/health",
  "/api/signals/inbox",
  "/api/om/intake",
  "/dri/enter",
]);

/**
 * 签名身份能进的地方。**不含 `/`、不含任何 `(tkos)` 下的页面** —— DRI 侧看不到
 * 一号位的提请台、对谈记录、判断本体。
 *
 * `/api/data` 在里面，但那不等于放开：能调哪个方法由 `METHOD_SCOPE` 说了算
 * （`src/lib/data/scope.ts`），而那一层查的是解出来的真身份。
 */
const DRI_PATHS = ["/dri", "/api/data", "/api/health"];

const inDriZone = (pathname: string): boolean =>
  DRI_PATHS.some((p) => pathname === p || pathname.startsWith(`${p}/`));

export function proxy(request: NextRequest) {
  const pathname = request.nextUrl.pathname;

  if (PUBLIC_PATHS.has(pathname)) return NextResponse.next();

  // 一号位那把口令开全站，照旧。两个 cookie 都带着的时候他说了算。
  if (isAccessCookieValid(request.cookies.get(ACCESS_COOKIE_NAME)?.value)) {
    return NextResponse.next();
  }

  /*
   * 签名身份只开 DRI 区。**这一层只验签，不认人** —— 「这个 personId 是不是 dri」
   * 要拿去对象库里解，而把 store.ts（连着整份种子）import 进 proxy 会把种子拖进
   * 这一层的 bundle。角色在 `/api/data` 与页面的服务端那一侧认（spec §4.4）。
   */
  const signed = personIdFromToken(request.cookies.get(WHO_COOKIE_NAME)?.value);
  if (signed && inDriZone(pathname)) return NextResponse.next();

  if (pathname.startsWith("/api/")) {
    return NextResponse.json({ error: "unauthorized" }, { status: 401 });
  }

  /*
   * 带着有效签名却敲了一号位的地址：**不要把他弹到 /access**。那一页要的是共享
   * 口令，他没有，于是他会以为自己的链接坏了，然后来问。照实说这一侧没有这一页。
   */
  if (signed) {
    return NextResponse.redirect(new URL("/dri", request.url));
  }

  const destination = new URL("/access", request.url);
  destination.searchParams.set("next", `${pathname}${request.nextUrl.search}`);
  return NextResponse.redirect(destination);
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|favicon.ico|icon.svg|brand/).*)"],
```

- [ ] **Step 2: 写 `/dri/enter`**

`src/app/dri/enter/route.ts`：

```ts
/**
 * 链接换 cookie。**这是 DRI 进来的唯一入口。**
 *
 * 是一个 GET route 而不是一个页面：cookie 只能在 Route Handler 或 Server Action
 * 里写（页面渲染期写不了），而这条链接是在群里被点开的，本来就是一次 GET。
 * 落 cookie 之后立刻 302 到 `/dri`，于是那条带签名的 URL 不会留在地址栏里，也不会
 * 被顺手转发出去。
 *
 * 与 `api/access/route.ts` 同一套 cookie 参数，包括 `secure` 靠
 * `x-forwarded-proto` 现判 —— 本地是 http，硬带 secure 的话浏览器根本不存，
 * 于是「进去了」之后一刷新又被弹回来。
 */
import { NextResponse } from "next/server";

import { WHO_COOKIE_MAX_AGE, WHO_COOKIE_NAME, personIdFromToken } from "@/lib/identity";

export const dynamic = "force-dynamic";
export const runtime = "nodejs";

const plain = (text: string, status: number) =>
  new NextResponse(text, {
    status,
    headers: { "content-type": "text/plain; charset=utf-8" },
  });

export async function GET(request: Request) {
  const token = new URL(request.url).searchParams.get("t") ?? undefined;
  const personId = personIdFromToken(token);

  if (!personId) {
    /*
     * 不重定向到任何地方 —— 他没有第二道门，弹到 /access 只会让他去猜一个口令。
     * 照实说链接不对，并说清楚下一步找谁。
     */
    return plain("这条链接无效或已过期。找刘明华重发一条。", 401);
  }

  const response = NextResponse.redirect(new URL("/dri", request.url));
  response.cookies.set({
    name: WHO_COOKIE_NAME,
    value: token!,
    httpOnly: true,
    secure: request.headers.get("x-forwarded-proto") === "https",
    sameSite: "lax",
    path: "/",
    maxAge: WHO_COOKIE_MAX_AGE,
    priority: "high",
  });
  return response;
}
```

```ts
 * 待挑的信号：**两种来源，共享一张待挑页。**
 *
 * 抓取作业（`tkos-lab/scripts/catch.py`）投进来的那一支可信来自「有人在外面盯着

 * 缺的字段不给，不给 0。

  /**
   * 幂等键。抓取那一支 = 归一化后的 URL（去 utm 与 fragment，作业那边由
   * `sources.normalize_url` 算）；人报那一支有 URL 就同上，没有就取
   * 「姓名 + 一句话 + 日期」的哈希（`src/lib/data/store.ts` 的 `reportedKey`）。

  title: string;
  /** 抓不到就是空串。人报的「出处」是一句话时也可以为空 —— 空串照样收。 */
  url: string;
  /** 正文或摘要。这边不再加工，只截长度。 */
  text: string;
  /** ISO 8601。抓不到就是空串，不猜一个时间填上去。 */
  publishedAt: string;
  /** 源的备注名或域名；人报的那一支写他填的「出处」。 */
  source: string;
  /** 抓取那一支是作业算的；人报那一支是「为什么是现在」。 */
  why: string;
}

/** 抓取作业独有的那七个字段。**这一侧不做任何研判**，只展示与排序。 */
export interface CrawledSignalInput extends InboxSignalCommon {
  kind: "crawled";
  origin: "pinned" | "discovered";
  /** 1–5，作业那边按 rubric 打的。**区间不在这边校验**（见 inbox.ts 里的注释）。 */
  score: number;
  dimensions: Record<string, number>;
  /** 命中的关键情报问题编号，例如 `KIQ2.2`。它是「为什么值得看」的可追溯锚。 */
  kiq: string;
  suggestedIssue: string;

  runId: string;
  scannedAt: string;
}

/** 一个 DRI 报上来的。四个字段，没有分数。 */
export interface ReportedSignalInput extends InboxSignalCommon {
  kind: "reported";
  /** 谁报的，姓名。这一支的全部可信度在这三个字上。 */
  by: string;
  /**
   * 他觉得这条动摇哪条判断（判断 id）。**允许为空是刻意的**：要求上报人先答出
   * 「动摇哪条判断」等于把上报门槛提到没人愿意报。那道门槛属于出卡侧，不属于收件侧。
   */
  shakes?: string;
}

export type InboxSignalInput = CrawledSignalInput | ReportedSignalInput;

/** 一号位对一条待挑信号做的处置。`issue` = 开成议题（去跑 A1a）。 */
export type InboxDisposition = "issue" | "watch" | "dismiss";

/** 落库之后多出来的那几样。两支共用。 */
export interface InboxSignalMeta {
  /** 收件时间。同一个 `dedupeKey` 再来一次**不改**它。 */
  receivedAt: string;
  /** 空 = 还在等他挑。**v1 只记不读**（rollout §7）。 */

  judged?: {
    runId: string;
    status: AgentRunRecord["status"];
    route?: AgentRunRecord["route"];
    at: string;
  };
}

export type InboxSignal =
  | (CrawledSignalInput & InboxSignalMeta)
  | (ReportedSignalInput & InboxSignalMeta);
```

```ts
/**
 * 人报那一条的幂等键。
 *
 * 有 URL 就用归一化 URL —— 两个人看到同一篇文章各报一次，那是同一条线索，
 * 不该在待挑页上出现两遍。没有 URL（「客户在电话里说的」）就取
 * 「姓名 + 一句话 + 日期」的哈希：同一个人同一天把同一句话再提一次是手滑，
 * 换个人提或者隔一天再提是新的信息。
 */
export function reportedKey(by: string, title: string, day: string): string {
  return `dri-${createHash("sha1").update(`${by}|${title}|${day}`).digest("hex").slice(0, 16)}`;
}
```

URL 归一化**用现成的那个**：`src/lib/scan/normalize-url.ts` 的 `normalizeUrl(url: string): string`（它和抓取作业那边 `sources.normalize_url` 的口径是对齐过的，还带自己的测试 `normalize-url.test.ts`）。在 `store.ts` 里 `import { normalizeUrl } from "@/lib/scan/normalize-url";` —— **不要再写一份**，两份口径迟早会分叉，而分叉的后果是「同一篇文章在待挑页上出现两遍」。

```ts
/**
 * 一条人报的信号落库。**同一个池子**（`store.inboxSignals`）——
 * `CardSource` 上 `{kind:"dri", by}` 那一档早就在了，这里补上的是它的来路。
 *
 * 第二次报同一条：什么都不改，把原来那条原样返回。**尤其不改 `receivedAt`**，
 * 也不复活已处置的那些 —— 理由同 `inboxSignalsPut`。
 */
export function reportSignal(input: ReportedSignalInput): InboxSignal {
  const cur = store.inboxSignals.find((s) => s.dedupeKey === input.dedupeKey);
  if (cur) return clone(cur);

  const put: InboxSignal = { ...input, receivedAt: nowIso() };
  store.inboxSignals.push(put);

  const doomed = inboxEvictees(store.inboxSignals, MAX_INBOX_SIGNALS);
  if (doomed.length > 0) {
    const keys = new Set(doomed.map((s) => s.dedupeKey));
    for (const s of doomed) {
      if (!s.disposedAs) {
        console.warn(
          `[tkos:inbox] 上限 ${MAX_INBOX_SIGNALS} 满了，挤掉一条**还没有人挑过**的信号：${s.title}。`
        );
      }
    }
    store.inboxSignals = store.inboxSignals.filter((s) => !keys.has(s.dedupeKey));
  }

  persistSoon();
  return clone(put);
}
```

（`createHash` 从 `node:crypto` import；`nowIso` / `clone` / `MAX_INBOX_SIGNALS` / `inboxEvictees` 都已在这个文件里。）

```tsx
 * 上报信号。一屏一个表单，四个字段。
 *
 * 「动摇哪条判断」**允许留空**，而且默认就是空：要求上报人先答出这个问题等于把

export function DriReportView() {
  const judgments = useResource("dri-judgments", () => api.listMyDomainJudgments());
  const mine = useResource<DriReportedSignal[]>("dri-reports", () => api.listMyReports());
  const invalidate = useInvalidate();

  const [title, setTitle] = useState("");
  const [source, setSource] = useState("");
  const [text, setText] = useState("");
  const [why, setWhy] = useState("");
  const [shakes, setShakes] = useState("");
  const [url, setUrl] = useState("");
  const [pending, setPending] = useState(false);

      setSource("");
      setText("");

    }
  };

      <PanelHead
        eyebrow="上报信号"
        title="你看到了什么"

/** 处置成什么了。**没被挑过就说没被挑过**，不写「处理中」——那会像在等一个进度。 */
function disposalWord(s: DriReportedSignal): string {
  if (!s.disposedAs) return "他还没挑到";
  if (s.disposedAs === "issue") return "已开成议题";
  if (s.disposedAs === "watch") return "他先放进待定";
  return "他看过，没往下走";
}

  children: React.ReactNode;
}) {

      <span className="text-base text-ink">{label}</span>
      <span className="meta mt-0.5 block text-ink-muted">{hint}</span>
      <span className="mt-2 block">{children}</span>
    </label>
  );
}
```

```ts
      keyEvidence: lines(doc.keyEvidence),
      counters: lines(doc.counters),
      unknowns: lines(doc.unknowns),
```

```ts
 *
 * **这是这一版唯一新建的本体对象。** `MissionCard` 是只读的（TKOS 里能看不能改），
 * 而回传是一个真正的写路径：公司提案一条 Mission，DRI 接受/修改/重写/拒绝并给出
 * 成功证据与里程碑，一号位逐条裁决，收敛之后一次生效。
 *
 * 8 月那次复盘认定「两轮收敛」的主因就是**条目级回传**（G2）—— 任何 IM 卡片承载
 * 不了这个结构，这也是这一侧要有网页的原因。

  /** `${domainId}:${periodId}` —— 一个域一期只有一份。 */
  id: string;
  domainId: string;
  periodId: string;
  version: number;
  /**
   * 纲领灌进来那一刻是 `drafted`（提案在了，DRI 还没标注）。

   * DRI 按裁决修订再提交 → version+1 且回到 `submitted`（这就是「一轮」）；
   * 一号位签一次 → `effective`，此后只读。
   */

  assumptions: { text: string; defaultPath: string; dueOn?: string }[];
  /** 生效前提，硬上限 2 条（M2 承诺最低条件）。 */
  preconditions: string[];
  /** 复盘 H2 那张表：开放问题不追踪就会静默过期。 */
  questions: { text: string; toWhom: string; needBy: string; closed: boolean }[];
  /**
   * 每一次提交落一条。与 `JudgmentRevision` 同形状，不另起一套。
   *
   * **收敛轮次是从这里数出来的，不设字段** —— 屏幕上每个数字都是算出来的
   * （`store.ts` 文件头第四条）。
   */
  revisions: { version: number; at: string; actor: string; note: string }[];
}

  id: string;
  /** 公司提案的那条 Mission，原文照录。改写它就是越界。 */

  /** 非 accept 必填。一个没有理由的「重写」在裁决时无法处理。 */
  reason?: string;
  /** 成功证据 3–5 项：行为与结果证据，不是功能清单。 */
  evidence: string[];
```

- [ ] **Step 4: 写状态机**

`src/lib/data/commitment.ts`：

```ts
import { ApiError } from "./shared";
import type { Commitment, CommitmentItem } from "./types";

/**
 * Commitment 的状态机与派生量。**纯函数：不碰 store、不读时钟、不抄引用。**
 *
 * 分出这个文件的理由是它是这一版唯一有状态机的东西，而状态机的每一条边都值得
 * 一个用例（spec §10）。时间与操作人由调用方传进来，于是测试里的时间线是确定的。
 */

type Stamp = { actor: string; at: string };

/** 生效之后只读。**一号位签过的东西不能被谁再改一笔。** */
export function assertWritable(c: Commitment): void {
  if (c.status === "effective") {
    throw new ApiError("这份承诺已经生效，改不了了。要改就开下一期。");
  }
}

/**
 * DRI 提交回传。`drafted`（第一次）或 `adjudicated`（按裁决修订）才能提交。
 *
 * 从 `adjudicated` 提交时 version+1 —— 一次「提交 + 裁决」就是复盘里说的一轮，
 * 8 月实测两轮。
 */
export function submitDraft(c: Commitment, opts: Stamp & { note: string }): Commitment {
  assertWritable(c);
  if (c.status === "submitted") {
    throw new ApiError("这份回传已经交出去了，等他裁完再改。");
  }

  const bad = c.items.find((i) => i.verdict !== "accept" && !i.reason?.trim());
  if (bad) {
    throw new ApiError(`「${bad.proposed}」标成了${verdictWord(bad.verdict)}，得写一句理由。`);
  }

  const version = c.status === "adjudicated" ? c.version + 1 : c.version;
  return {
    ...c,
    status: "submitted",
    version,
    revisions: [
      ...c.revisions,
      { version, at: opts.at, actor: opts.actor, note: opts.note.trim() },
    ],
  };
}

/**
 * 一号位逐条裁决。裁决落在条目上（各带 `at`/`by`），不另存一份裁决记录 ——
 * 「这一条是谁什么时候怎么裁的」必须跟着那一条走。
 *
 * 只裁一部分是允许的：没裁的那几条 `ruling` 保持原样，DRI 那侧照实显示「他还没裁」。
 */
export function adjudicate(
  c: Commitment,
  rulings: Record<string, { verdict: "endorse" | "revise" | "reject"; note: string }>,
  opts: Stamp
): Commitment {
  assertWritable(c);
  if (c.status !== "submitted") {
    throw new ApiError("这个域还没有回传，没有东西可裁。");
  }

  const known = new Set(c.items.map((i) => i.id));
  const unknown = Object.keys(rulings).filter((id) => !known.has(id));
  if (unknown.length > 0) {
    // 静默忽略等于「他点了裁决，屏幕上什么也没变」——那会被读成系统坏了。
    throw new ApiError(`裁决里有不存在的条目：${unknown.join("、")}。`);
  }

  return {
    ...c,
    status: "adjudicated",
    items: c.items.map((item) => {
      const r = rulings[item.id];
      return r ? { ...item, ruling: { ...r, note: r.note.trim(), at: opts.at, by: opts.actor } } : item;
    }),
  };
}

/** 一号位签一次，生效。此后只读。 */
export function makeEffective(c: Commitment, opts: Stamp): Commitment {
  assertWritable(c);
  if (c.status === "drafted") {
    throw new ApiError("这个域还没有回传，签不了。");
  }
  return {
    ...c,
    status: "effective",
    revisions: [
      ...c.revisions,
      { version: c.version, at: opts.at, actor: opts.actor, note: "签发生效" },
    ],
  };
}

/** 收敛轮次 = 提交过几次。**算出来的，不存字段。** */
export function convergenceRounds(c: Commitment): number {
  return c.revisions.filter((r) => r.note !== "签发生效").length;
}

/**
 * 「修改」超过两条时的那句话。
 *
 * **不拦提交。** 这是复盘 G3 的经验 —— 分歧超过两处通常是结构性的，先约谈比再写
 * 一版便宜。把经验做成硬校验就变成官僚：他有可能真的就是要改五条。
 */
export function disagreementNote(c: Commitment): string | null {
  const n = c.items.filter((i) => i.verdict === "modify" || i.verdict === "rewrite").length;
  if (n <= 2) return null;
  return `这一份里有 ${n} 条改了提案。分歧超过两处通常是结构性的 —— 先约一次谈比再回传一版便宜。`;
}

const verdictWord = (v: CommitmentItem["verdict"]): string =>
  v === "modify" ? "修改" : v === "rewrite" ? "重写" : "拒绝";
```

落库函数：

```ts
/**
 * 灌纲领。**幂等，而且只覆盖还没人动过的那些。**
 *
 * 同一期同一个域重复灌：`drafted`（DRI 还没标注）就整条换掉 —— 纲领改了一版，
 * 重灌一次是正常操作。已经 `submitted`/`adjudicated`/`effective` 的**跳过并报出来**：
 * 那意味着有人已经在上面写了东西，静默覆盖等于把一个人半天的回传抹掉。
 */
export function commitmentsIntake(payload: OmIntake): {
  periodCreated: boolean;
  drafted: number;
  skipped: string[];
} {
  let periodCreated = false;
  if (!store.periods.some((p) => p.id === payload.period.id)) {
    store.periods.push({ ...payload.period, projectId: tk.projects[0].id });
    periodCreated = true;
  }
  if (!store.companyOutcomes.some((o) => o.periodId === payload.period.id)) {
    store.companyOutcomes.push({
      id: `CO-${payload.period.id}`,
      periodId: payload.period.id,
      statement: payload.companyOutcome.statement,
      dueOn: payload.companyOutcome.dueOn,
    });
  }

  let drafted = 0;
  const skipped: string[] = [];
  for (const d of payload.domains) {
    if (!allDomains.some((x) => x.id === d.domainId)) {
      skipped.push(`${d.domainId}（本体里没有这个域）`);
      continue;
    }
    const id = `${d.domainId}:${payload.period.id}`;
    const cur = store.commitments.find((c) => c.id === id);
    if (cur && cur.status !== "drafted") {
      skipped.push(`${d.domainId}（已经是 ${cur.status}，没动）`);
      continue;
    }
    const made: Commitment = {
      id,
      domainId: d.domainId,
      periodId: payload.period.id,
      version: 1,
      status: "drafted",
      northStar: d.northStar,
      items: d.items.map((it, n) => ({
        id: `${id}-m${n + 1}`,
        proposed: it.proposed,
        /* 初值是 accept：纲领刚下来的时候没有分歧，分歧是 DRI 标出来的。 */
        verdict: "accept",
        evidence: [],
        milestone: {},
        tier: it.tier,
      })),
      assumptions: [],
      preconditions: [],
      questions: [],
      revisions: [],
    };
    if (cur) store.commitments[store.commitments.indexOf(cur)] = made;
    else store.commitments.push(made);
    drafted += 1;
  }

  persistSoon();
  return { periodCreated, drafted, skipped };
}
```

```bash
#!/usr/bin/env bash
# .data/store.json 带日期拷一份。一行 crontab 的那个「一行」。

#   0 3 * * * /srv/tkos/scripts/backup-store.sh >> /var/log/tkos-backup.log 2>&1

DEST="${TKOS_BACKUP_DIR:-/srv/tkos/backups}"
KEEP="${TKOS_BACKUP_KEEP:-30}"

if [ ! -f "$SRC" ]; then
  echo "[tkos:backup] 没有 $SRC —— 应用还没写过盘，或者路径配错了。"
  exit 1
fi

mkdir -p "$DEST"
STAMP="$(date +%Y%m%d-%H%M%S)"
cp "$SRC" "$DEST/store-$STAMP.json"
echo "[tkos:backup] $DEST/store-$STAMP.json（$(wc -c <"$SRC") 字节）"

# 只留最近 KEEP 份。`ls -t` 按时间倒序，tail 掉前 KEEP 个之后的全删。
ls -t "$DEST"/store-*.json | tail -n "+$((KEEP + 1))" | xargs -r rm --
```
